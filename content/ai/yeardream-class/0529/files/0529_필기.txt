weight    = 80.0      # kg — 원하는 값으로 변경
height_cm = 181.0     # cm — 원하는 값으로 변경

# bmi 공식
height_m = height_cm / 100
# bmi      = weight / (height_m * height_m) 
bmi      = weight / (height_m ** 2)
print("BMI:", round(bmi, 2)) 

# 분류표 ==> 의료쪽에서 제공하는 정보 (정보 탐색 능력)
if bmi < 18.5:
    print("저체중입니다.")
elif bmi < 23:
    print("정상 체중입니다.")
elif bmi < 25:
    print("과체중입니다.")
else:
    print("비만입니다.")
