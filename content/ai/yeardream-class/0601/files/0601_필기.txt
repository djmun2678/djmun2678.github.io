# 아래 성적 딕셔너리를 분석하세요
grades = {
    "수학": 95,
    "영어": 88,
    "과학": 92,
    "국어": 79,
    "역사": 85
}

# 1) 전체 평균 구하기
total = 0
for subject, score in grades.items():
    total += score
average = total / len(grades)
print(f"평균: {average:.1f}")

# 2) 90점 이상인 과목만 출력하기
print("90점 이상 과목:")
for subject, score in grades.items():
    if score >= 90:         # 90점 이상일 때만 출력
        print(f"  - {subject}: {score}점")


# return 없을 때 → 함수 밖에서 결과를 사용할 수 없음
def formula_no_return(a, b):
    c = (a**2) + (b**2)
    print("함수 안:", c)     # 출력은 되지만...

result = formula_no_return(2, 4)
print("함수 밖:", result)    # None ← 반환값이 없으면 None!

print()

# return 있을 때 → 결과를 변수에 저장 가능
def formula_with_return(a, b):
    c = (a**2) + (b**2)
    return c                 # 값을 반환!

result = formula_with_return(2, 4)
print("함수 밖:", result)    # 20 ← 정상적으로 값을 받음
